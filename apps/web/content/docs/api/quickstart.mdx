You need a Workspace API key with the `processing:write` and `exports:write` permissions. See [Authentication](/docs/api/introduction#authentication).

## Set up

<CodeGroup>

```typescript TypeScript
// Node 18 or newer (or Bun). Run it with NARRIFLOW_API_KEY set.
const BASE = "https://app.narriflow.com/api/v1";

type Options = Omit<RequestInit, "body"> & { idempotencyKey?: string; body?: unknown };

async function api<T>(path: string, { idempotencyKey, body, ...init }: Options = {}): Promise<T> {
  const response = await fetch(BASE + path, {
    ...init,
    headers: {
      Authorization: `Bearer ${process.env.NARRIFLOW_API_KEY}`,
      "Content-Type": "application/json",
      ...(idempotencyKey && { "Idempotency-Key": idempotencyKey }),
    },
    body: body === undefined ? undefined : JSON.stringify(body),
  });
  const payload = (await response.json()) as T & { error?: string; message?: string; requestId?: string };
  if (!response.ok) throw new Error(`${payload.error}: ${payload.message} (${payload.requestId})`);
  return payload;
}

interface Operation {
  operationId: string;
  state: "working" | "completed" | "partial" | "failed" | "cancelled";
  results: { clipIds: string[] };
}

async function waitFor(operationId: string): Promise<Operation> {
  for (;;) {
    const status = await api<Operation>(`/operations/${operationId}`);
    if (status.state !== "working") return status;
    await new Promise((resolve) => setTimeout(resolve, 15_000));
  }
}
```

```python Python
# Needs the requests package. Run it with NARRIFLOW_API_KEY set.
import os
import time

import requests

BASE = "https://app.narriflow.com/api/v1"
session = requests.Session()
session.headers["Authorization"] = f"Bearer {os.environ['NARRIFLOW_API_KEY']}"


def api(method, path, idempotency_key=None, **kwargs):
    headers = {"Idempotency-Key": idempotency_key} if idempotency_key else {}
    response = session.request(method, BASE + path, headers=headers, timeout=30, **kwargs)
    payload = response.json()
    if not response.ok:
        raise RuntimeError(f"{payload['error']}: {payload['message']} ({payload['requestId']})")
    return payload


def wait_for(operation_id):
    while True:
        status = api("GET", f"/operations/{operation_id}")
        if status["state"] != "working":
            return status
        time.sleep(15)
```

## Make the requests

<Steps>
  <Step title="Check your key">
    `GET /workspace` returns the Workspace the key belongs to, its plan, and the processing minutes you have left.

    <CodeGroup>

    ```typescript TypeScript
    const { workspace, usage } = await api<{
      workspace: { name: string };
      usage: { remainingMinutes: number };
    }>("/workspace");
    console.log(workspace.name, usage.remainingMinutes);
    ```

    ```bash cURL
    curl "$BASE/workspace" -H "Authorization: Bearer $NARRIFLOW_API_KEY"
    ```

    ```python Python
    info = api("GET", "/workspace")
    print(info["workspace"]["name"], info["usage"]["remainingMinutes"])
    ```

    </CodeGroup>
  </Step>
  <Step title="Submit a video link">
    `POST /projects` imports a link and starts clip generation. It returns `202` with an operation to follow. Use a new `Idempotency-Key` for each new video. The link must come from YouTube, Google Drive, StreamYard, Loom, Twitch, X, TikTok, LinkedIn, Facebook, Vimeo or Dropbox.

    <CodeGroup>

    ```typescript TypeScript
    const submitted = await api<{ projectId: string; operation: { operationId: string } }>("/projects", {
      method: "POST",
      idempotencyKey: "episode-412:submit",
      body: { url: "https://www.youtube.com/watch?v=VIDEO_ID", generation: { clipCountTarget: 6 } },
    });
    ```

    ```bash cURL
    curl -X POST "$BASE/projects" \
      -H "Authorization: Bearer $NARRIFLOW_API_KEY" \
      -H "Idempotency-Key: episode-412:submit" \
      -H "Content-Type: application/json" \
      -d '{"url": "https://www.youtube.com/watch?v=VIDEO_ID", "generation": {"clipCountTarget": 6}}'
    ```

    ```python Python
    submitted = api(
        "POST",
        "/projects",
        idempotency_key="episode-412:submit",
        json={"url": "https://www.youtube.com/watch?v=VIDEO_ID", "generation": {"clipCountTarget": 6}},
    )
    ```

    </CodeGroup>

    Save `operation.operationId` and `projectId` from the response.
  </Step>
  <Step title="Wait for the clips">
    Poll `GET /operations/{operationId}` until `state` is no longer `working`. The intake covers the import and the clip generation that follows it. `completed` means every clip succeeded, and `partial` means some failed.

    <CodeGroup>

    ```typescript TypeScript
    const intake = await waitFor(submitted.operation.operationId);
    console.log(intake.state, intake.results.clipIds.length);
    ```

    ```bash cURL
    until [ "$(curl -s "$BASE/operations/$OPERATION_ID" \
      -H "Authorization: Bearer $NARRIFLOW_API_KEY" | jq -r .state)" != "working" ]; do
      sleep 15
    done
    ```

    ```python Python
    intake = wait_for(submitted["operation"]["operationId"])
    print(intake["state"], len(intake["results"]["clipIds"]))
    ```

    </CodeGroup>
  </Step>
  <Step title="List the best clips">
    `GET /projects/{id}/clips` returns clips ranked by virality.

    <CodeGroup>

    ```typescript TypeScript
    interface Clip {
      clipId: string;
      editorRevision: number;
      title: string | null;
      scores: { virality: number };
    }

    const { items } = await api<{ items: Clip[] }>(`/projects/${submitted.projectId}/clips?limit=3`);
    const [clip] = items;
    if (!clip) throw new Error("The project has no clips yet");
    console.log(clip.title, clip.scores.virality);
    ```

    ```bash cURL
    curl "$BASE/projects/$PROJECT_ID/clips?limit=3" \
      -H "Authorization: Bearer $NARRIFLOW_API_KEY"
    ```

    ```python Python
    clip = api("GET", f"/projects/{submitted['projectId']}/clips", params={"limit": 3})["items"][0]
    print(clip["title"], clip["scores"]["virality"])
    ```

    </CodeGroup>

    Save the top clip's `clipId` and `editorRevision`.
  </Step>
  <Step title="Export a clip">
    `POST /clips/{id}/exports` exports an exact editor revision. Send the `editorRevision` you just read as `expectedRevision`. If someone edits the clip first, the request fails instead of exporting the wrong version.

    <CodeGroup>

    ```typescript TypeScript
    const created = await api<{ export: { exportId: string }; operation: { operationId: string } }>(
      `/clips/${clip.clipId}/exports`,
      {
        method: "POST",
        idempotencyKey: "episode-412:export:1",
        body: { expectedRevision: clip.editorRevision, aspectRatios: ["9:16"], resolution: "1080p" },
      },
    );
    await waitFor(created.operation.operationId);
    ```

    ```bash cURL
    curl -X POST "$BASE/clips/$CLIP_ID/exports" \
      -H "Authorization: Bearer $NARRIFLOW_API_KEY" \
      -H "Idempotency-Key: episode-412:export:1" \
      -H "Content-Type: application/json" \
      -d '{"expectedRevision": 0, "aspectRatios": ["9:16"], "resolution": "1080p"}'
    ```

    ```python Python
    created = api(
        "POST",
        f"/clips/{clip['clipId']}/exports",
        idempotency_key="episode-412:export:1",
        json={"expectedRevision": clip["editorRevision"], "aspectRatios": ["9:16"], "resolution": "1080p"},
    )
    wait_for(created["operation"]["operationId"])
    ```

    </CodeGroup>

    The response's `output` shows the resolution you requested and the one your plan granted, with a `reason` of `plan_limit` when they differ.
  </Step>
  <Step title="Download the files">
    `GET /exports/{id}/download` issues fresh download links that last 15 minutes, plus SRT and VTT subtitles. Call it again for new links.

    <CodeGroup>

    ```typescript TypeScript
    const files = await api<{ variants: Array<{ downloadUrl: string }> }>(
      `/exports/${created.export.exportId}/download`,
    );
    console.log(files.variants[0]?.downloadUrl);
    ```

    ```bash cURL
    curl "$BASE/exports/$EXPORT_ID/download" \
      -H "Authorization: Bearer $NARRIFLOW_API_KEY"
    ```

    ```python Python
    files = api("GET", f"/exports/{created['export']['exportId']}/download")
    print(files["variants"][0]["downloadUrl"])
    ```

    </CodeGroup>
  </Step>
</Steps>

Instead of polling, you can ask Narriflow to call your server when the work finishes. See [Completion callbacks](/docs/api/callbacks). To send a local file instead of a link, see [Uploads](/docs/api/uploads).
